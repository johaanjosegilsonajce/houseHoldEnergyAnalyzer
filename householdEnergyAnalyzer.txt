#include <stdio.h>

#define MAX 50                      /* room for 50 records */

int main(void)
{
    int month[MAX];                 /* month number 1-12 */
    float units[MAX];               /* units (kWh) used */
    int count = 0;                  /* records stored so far */
    int choice, i, m, found;
    float u;

    do
    {
        /* ---- menu ---- */
        printf("\n====================================\n");
        printf("   HOUSEHOLD ENERGY ANALYZER\n");
        printf("====================================\n");
        printf("  1. Add month\n");
        printf("  2. View all months\n");
        printf("  3. Search by month number\n");
        printf("  4. Exit\n");
        printf("------------------------------------\n");
        printf("Enter your choice: ");
        if (scanf("%d", &choice) != 1)
        {
            scanf("%s");           / discard bad input */
            choice = 0;
        }

        switch (choice)
        {
        case 1:                     /* Add month */
            if (count >= MAX)
            {
                printf("Storage full! Cannot add more records.\n");
                break;
            }
            printf("Enter month number (1-12): ");
            scanf("%d", &m);
            if (m < 1 || m > 12)
            {
                printf("Invalid month! Enter a number from 1 to 12.\n");
                break;
            }
            printf("Enter units used: ");
            scanf("%f", &u);
            if (u < 0)
            {
                printf("Invalid units! Units cannot be negative.\n");
                break;
            }
            month[count] = m;
            units[count] = u;
            count++;
            printf("Record added with S.No. %d.\n", count);
            break;

        case 2:                     /* View all months */
            if (count == 0)
            {
                printf("No records to display.\n");
                break;
            }
            printf("\n%-6s %-8s %-10s\n", "S.No.", "Month", "Units");
            printf("------------------------\n");
            for (i = 0; i < count; i++)
                printf("%-6d %-8d %-10.2f\n", i + 1, month[i], units[i]);
            break;

        case 3:                     /* Search by month number */
            printf("Enter month number to search: ");
            scanf("%d", &m);
            found = 0;
            for (i = 0; i < count; i++)
            {
                if (month[i] == m)
                {
                    if (found == 0)
                    {
                        printf("\n%-6s %-8s %-10s\n", "S.No.", "Month", "Units");
                        printf("------------------------\n");
                    }
                    printf("%-6d %-8d %-10.2f\n", i + 1, month[i], units[i]);
                    found = 1;
                }
            }
            if (found == 0)
                printf("No record found for month %d.\n", m);
            break;

        case 4:
            printf("Exiting program. Goodbye!\n");
            break;

        default:
            printf("Invalid choice! Please enter 1-4.\n");
        }
    } while (choice != 4);

    return 0;
}
